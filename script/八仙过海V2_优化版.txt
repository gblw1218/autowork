V2:=3;
V3:=5;
V4:=26;

DEA:=EMA(DIFF,V2);
A1:=DIFF>DEA;

RSV1:=(CLOSE-LLV(LOW,V3))/(HHV(HIGH,V3)-LLV(LOW,V3))*100;
K:=SMA(RSV1,3,1);
D:=SMA(K,3,1);
A2:=K>D;

CLOSEVAR:=REF(CLOSE,1);
RSI1:=(SMA(MAX(CLOSE-CLOSEVAR,0),V2,1))/(SMA(ABS(CLOSE-CLOSEVAR),V2,1))*100;
RSI2:=(SMA(MAX(CLOSE-CLOSEVAR,0),V4,1))/(SMA(ABS(CLOSE-CLOSEVAR),V4,1))*100;
A3:=RSI1>RSI2;


RSV_LWR:=-(HHV(HIGH,V4)-CLOSE)/(HHV(HIGH,V4)-LLV(LOW,V4))*100;
LWR1:=SMA(RSV_LWR,V1,1);
LWR2:=SMA(LWR1,V1,1);
A4:=LWR1>LWR2;


BBI:=(MA(CLOSE,V1)+MA(CLOSE,V2)+MA(CLOSE,V3)+MA(CLOSE,V4))/4;
A5:=CLOSE>BBI;

MTM:=CLOSE-REF(CLOSE,1);
MMS:=100*EMA(EMA(MTM,V2),V1)/EMA(EMA(ABS(MTM),V2),V1);
MMM:=100*EMA(EMA(MTM,V4),V3)/EMA(EMA(ABS(MTM),V4),V3);
A6:=MMS>MMM;

BIAS:=(C-MA(C,V2))/MA(C,V2);
DIF:=(BIAS-REF(BIAS,16));
DBCD:=SMA(DIF,76,1);
MM:=MA(DBCD,5);
A7:=DBCD>MM;

mtmb:=c-ref(c,1);
ZLGJ:=100*ema(ema(mtmb,V3),V3)/ema(ema(abs(mtmb),V3),V3);
MAZL:=MA(ZLGJ,5);
A8:=ZLGJ>MAZL;

状态:(A1+A2+A3+A4+A5+A6+A7+A8),linethick2,COLORBLUE;
量能比:=VOL/MA(VOL,5);
操作建议:=IF(状态>=7,'坚决多',IF(状态>=5,'试多',IF(状态<=1,'坚决空',IF(状态<=3,'试空','观望'))));

DRAWTEXT(量能比>1.5 AND ISLASTBAR=1, 9, '量能:放量'),COLORRED;
DRAWTEXT(量能比<0.7 AND ISLASTBAR=1, 9, '量能:缩量'),COLORGREEN;
DRAWTEXT(量能比>=0.7 AND 量能比<=1.5 AND ISLASTBAR=1, 9, '量能:平量'),COLORWHITE;

DRAWTEXT(状态>=7 AND ISLASTBAR=1, 9.3, '操作:坚决多'),COLORRED;
DRAWTEXT(状态>=5 AND 状态<7 AND ISLASTBAR=1, 9.3, '操作:试多'),COLORBLUE;
DRAWTEXT(状态<=1 AND ISLASTBAR=1, 9.3, '操作:坚决空'),COLORGREEN;
DRAWTEXT(状态<=3 AND 状态>1 AND ISLASTBAR=1, 9.3, '操作:试空'),COLORBLUE;
DRAWTEXT(状态>3 AND 状态<5 AND ISLASTBAR=1, 9.3, '操作:观望'),COLORWHITE;

波幅:MA(TR1,NVAL),NODRAW,COLOR4B3CC9;
